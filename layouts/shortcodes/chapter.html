{{/* chapter — 引用小说章节，自动取到标题与永久链接。样式用原子类。 用法： {{< chapter 12 >}} ← 本书（沿用当前页的 novel 字段） {{< chapter novel="sky-tax" n="12" >}} ← 跨书引用 未找到时：输出「第 N 章」占位并 warnf —— 本站构建带 --panicOnWarning， 写错章节号会当场失败，而不是留一个静默的死链。 */}} {{- $novel := .Get "novel" -}} {{- if not $novel }}{{ $novel = .Page.Params.novel }}{{ end -}} {{- $n := .Get "n" | default (.Get 0) -}} {{- $target := "" -}} {{- with $novel -}} {{- range (where site.RegularPages "Params.novel" .) -}} {{- if eq (int (.Params.chapter | default 0)) (int $n) -}}{{- $target = . -}}{{- end -}} {{- end -}} {{- end -}} {{- with $target -}} {{ .Title }} {{- else -}} {{- warnf "shortcode chapter: 找不到 novel=%q chapter=%v（页面 %s）" $novel $n $.Page.File.Path -}} 第 {{ $n }} 章 {{- end -}}
