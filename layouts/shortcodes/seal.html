{{/* seal — 朱红印章。样式是原子类 + 两个任意属性（Tailwind 没有 writing-mode 的现成 工具类，用 [writing-mode:vertical-rl] / [text-orientation:upright] 表达）。 用法： {{< seal >}}亦幸{{< /seal >}} {{< seal size="sm" >}}阁{{< /seal >}} {{< seal shape="round" size="lg" >}}幸{{< /seal >}} 参数：size（sm / md / lg）、shape（square / round）。 无障碍：role="img" + aria-label，读屏念「印：亦幸」。 */}} {{- $text := .Inner | plainify | strings.TrimSpace -}} {{- $size := .Get "size" | default "md" -}} {{- $shape := .Get "shape" | default "square" -}} {{- $sizeClass := "text-base" -}} {{- if eq $size "sm" }}{{ $sizeClass = "text-[0.8rem]" }}{{ else if eq $size "lg" }}{{ $sizeClass = "text-[1.35rem]" }}{{ end -}} {{- $shapeClass := cond (eq $shape "round") "rounded-full px-[0.45em] py-[0.5em]" "rounded-sm px-[0.25em] py-[0.35em]" -}} {{- with $text -}} {{ . }} {{- end -}}
